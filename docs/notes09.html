Threads: counter.c (skit)¶

Question: Is the version without locks safe/correct?

Skit¶

Two people come up:

  • Person A (I/O): Listens for user's command
  • Person B (CPU): Updates counter
  • Scenario: What if there is only one CPU... and Person B gets interrupted before updating... race condition!

We need atomicity! Region of code with shared resource is critical section... before accessing shared resource, we need to grab mutex/lock!

  • Redo skit with people grabbing lock before accessing shared resource.

  • If someone has the lock, other threads will voluntarily wait until the lock is free.

  • Threads can choose to ignore lock!

We should abstract the lock away (Tuesday).

Locks: POSIX Threads¶

Use a lock or mutual exclusion to guard a critical section, which is a region of code that acces a shared resource:

pthread_mutex_t Lock;               // Declare lock
pthread_mutex_init(&Lock, NULL);    // Initialize lock

pthread_mutex_lock(&Lock);          // Acquire lock
access_shared_resource();           // Perform critical section
pthread_mutex_unlock(&Lock);        // Release lock

Threads: counter.c¶

/* Globals */
int         Counter  = 0;
pthread_mutex_t Lock = PTHREAD_MUTEX_INITIALIZER;

/* Functions */

void *update_counter(void *arg) {   // Background thread
    while (true) {                  // Version 0: Just count (no loop)
        usleep(TIMEOUT * 1000);
        pthread_mutex_lock(&Lock);  // Version 1: No lock
        Counter += 1;
        pthread_mutex_unlock(&Lock);
    }

    return NULL;
}

/* Main execution */

int main(int argc, char *argv[]) {
    // Background Thread - Update Loop
    pthread_t thread;
    pthread_create(&thread, NULL, update_counter, 0);

    // Foreground Thread - I/O Loop
    while (true) {
        char buffer[BUFSIZ];

        // Display prompt
        fputs(PROMPT, stdout); fflush(stdout);

        // Read input
        if (!fgets(buffer, BUFSIZ, stdin)) {
            break;
        }
        buffer[strlen(buffer) - 1] = 0;

        // Process input
        if (strcmp(buffer, "count") == 0) {
            pthread_mutex_lock(&Lock);    // Version 1: No locks
            fprintf(stdout, "Counter is %d\n", Counter);
            pthread_mutex_unlock(&Lock);
        } else if (strcmp(buffer, "reset") == 0) {
            pthread_mutex_lock(&Lock);    // Version 1: No locks
            Counter = 0;
            pthread_mutex_unlock(&Lock);
            fputs("Counter is reset!\n", stdout);
        } else if (strcmp(buffer, "exit") == 0) {
            fputs("Bye!\n", stdout);
            break;
        } else {
            fprintf(stderr, "Unknown command: %s\n", buffer);
        }
    }

    return 0;
}

Locks: prime (single-threaded)¶

size_t is_prime(size_t n) {
    for (size_t i = 2; i < n; i++) {
        if (n % i == 0)
            return 0;
    }
    return 1;
}

int main(int argc, char *argv[]) {
    size_t count = 0;
    for (size_t i = 2; i < PRIME_MAX; i++) {
        count += is_prime(i);
    }
    printf("There are %lu primes less than %lu\n", count, PRIME_MAX);
    return 0;
}

Questions:

  • Is this problem data or task parallel? Data parallel.

  • How could I make this faster? Use threads to process ranges.

Locks: prime (multi-threaded - v1)¶

/* Structure */
typedef struct {
    size_t start;
    size_t stop;
} PrimeArgs;

// Version 0: No locks
void * count_primes(void *arg) {
    PrimeArgs *pa = (PrimeArgs *)arg;
    printf("start=%lu, stop=%lu\n", pa->start, pa->stop);

    for (size_t n = pa->start; n < pa->stop; n++) {
        if (is_prime(n)) {
            Count++;
        }
    }
    return NULL;
}

int main(int argc, char *argv[]) {
    pthread_t thread;
    PrimeArgs args;

    // Division of work
    for (size_t t = 0; t < NUM_THREADS; t++) {
        args.start = max(2, t*PRIME_MAX/NUM_THREADS);
        args.stop  = (t+1)*PRIME_MAX/NUM_THREADS;
        pthread_create(&threads, NULL, count_primes, &args);
    }

    printf("There are %lu primes less than %ld\n", Count, PRIME_MAX);
    return 0;
}

Observation:

  • Problem 1: If main thread exits... everything exits.

  • Solution: Wait for worker threads.

Locks: prime (multi-threaded - v2)¶

int main(int argc, char *argv[]) {
    pthread_t threads[NUM_THREADS]  // Need array of threads
    PrimeArgs args;

    // Division of work
    for (size_t t = 0; t < NUM_THREADS; t++) {
        args.start = max(2, t*PRIME_MAX/NUM_THREADS);
        args.stop  = (t+1)*PRIME_MAX/NUM_THREADS;
        pthread_create(&threads[t], NULL, count_primes, &args);
    }

    // Wait for threads
    for (size_t t = 0; t < NUM_THREADS; t++) {
        pthread_join(threads[t], NULL);

}
